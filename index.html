diffusion_example.py
from maldatagen.models.diffusion import DiffusionModel
from maldatagen.generators import DiffusionGenerator
import numpy as np

# Initialize diffusion model with custom parameters
model = DiffusionModel(
    steps=1000,
    hidden_dim=512,
    num_blocks=4,
    dropout=0.1,
    guidance_scale=7.5,
    noise_schedule="cosine"
)

# Create generator with advanced settings
generator = DiffusionGenerator(
    model=model,
    batch_size=256,
    sampling_method="ddim",
    acceleration=True
)

# Load and prepare your data
data = np.load("your_dataset.npy")
preprocessed_data = generator.preprocess(data)

# Train the model
generator.train(
    data=preprocessed_data,
    epochs=500,
    learning_rate=1e-4,
    validation_split=0.2,
    early_stopping=True
)

# Generate synthetic samples
synthetic = generator.generate(
    n_samples=10000,
    class_conditioning=True,
    temperature=0.7,
    seed=42
)

# Save results
generator.save("diffusion_model_weights.h5")
np.save("synthetic_data.npy", synthetic)

cgan_example.py
from maldatagen.models.cgan import CGAN
from maldatagen.generators import CGANGenerator
import pandas as pd

# Initialize CGAN with conditional generation
model = CGAN(
    latent_dim=100,
    num_classes=10,
    generator_layers=[256, 512, 1024],
    discriminator_layers=[1024, 512, 256],
    dropout=0.3,
    label_smoothing=True
)

# Create generator
generator = CGANGenerator(
    model=model,
    batch_size=128,
    label_type="one_hot"
)

# Load data with labels
data = pd.read_csv("labeled_data.csv")
features = data.drop("label", axis=1)
labels = data["label"]

# Train with class balancing
generator.train(
    X=features,
    y=labels,
    epochs=200,
    learning_rate=0.0002,
    class_weights="balanced",
    gradient_penalty=10.0
)

# Generate specific classes
class_distribution = {
    0: 1000,  # Class 0 samples
    1: 500,   # Class 1 samples
    2: 2000,  # Class 2 samples
}

synthetic_data = generator.generate(
    class_distribution=class_distribution,
    noise_seed=42,
    interpolate_classes=True
)

# Evaluate per-class quality
metrics = generator.evaluate_per_class(features, synthetic_data, labels)

wgan_example.py
from maldatagen.models.wgan import WGAN
from maldatagen.generators import WGANGenerator
import torch

# Initialize WGAN with gradient penalty
model = WGAN(
    latent_dim=128,
    critic_iterations=5,
    gradient_penalty_weight=10.0,
    weight_clipping=0.01,
    spectral_norm=True
)

# Create generator with WGAN-specific settings
generator = WGANGenerator(
    model=model,
    batch_size=64,
    n_critic=5,
    clip_value=0.01
)

# Prepare your data
data = torch.randn(10000, 50)  # Example data

# Train with improved stability
for epoch in range(500):
    for _ in range(5):  # Train critic multiple times
        critic_loss = generator.train_critic(data)

    generator_loss = generator.train_generator(data)

    # Log metrics
    if epoch % 50 == 0:
        wasserstein_distance = generator.calculate_wasserstein(data)
        print(f"Epoch {epoch}: WD = {wasserstein_distance:.4f}")

# Generate synthetic data
synthetic = generator.generate(
    n_samples=5000,
    truncation=0.7,  # Truncation trick for better quality
    seed=42
)

# Evaluate with Wasserstein distance
wd_score = generator.wasserstein_score(data, synthetic)
print(f"Wasserstein Distance: {wd_score:.4f}")

Training Stability

Eliminates mode collapse and vanishing gradients with Wasserstein distance.

Meaningful Loss

Wasserstein distance provides meaningful training signal and correlates with sample quality.

Gradient Penalty

Improved gradient penalty for Lipschitz constraint enforcement without weight clipping.

Mode Coverage

Better coverage of data distribution modes compared to standard GANs.

Model Examples

🎯 Conditional GAN (CGAN)

Generate synthetic samples using Conditional GANs with label conditioning for balanced datasets. Perfect for handling imbalanced data and controlling generation per class.

Class Balancing Conditional Generation Binary Classification 
import numpy
import tensorflow
from tensorflow.keras.optimizers import Adam
from tensorflow.keras.losses import BinaryCrossentropy
from tensorflow.keras.utils import to_categorical
from maldatagen.engine.models.Adversarial.AdversarialModel import AdversarialModel
from maldatagen.engine.algorithms.Adversarial.AdversarialAlgorithm import AdversarialAlgorithm

# Configuration
number_samples_per_class = {
    "classes": {0: 100, 1: 200, 2: 150},
    "number_classes": 3
}
input_shape = 1200

# Initialize Adversarial Model
adversarial_model = AdversarialModel(
    latent_dimension=100,
    output_shape=input_shape,
    activation_function='leakyrelu',
    initializer_mean=0.0,
    initializer_deviation=0.02,
    dropout_decay_rate_g=0.3,
    dropout_decay_rate_d=0.4,
    last_layer_activation='tanh',
    dense_layer_sizes_g=[256, 512, 1024],
    dense_layer_sizes_d=[1024, 512, 256],
    dataset_type=numpy.float32,
    number_samples_per_class=number_samples_per_class
)

# Setup algorithm
adversarial_algorithm = AdversarialAlgorithm(
    generator_model=adversarial_model.get_generator(),
    discriminator_model=adversarial_model.get_discriminator(),
    latent_dimension=100,
    loss_generator='binary_crossentropy',
    loss_discriminator='binary_crossentropy',
    file_name_discriminator='discriminator.h5',
    file_name_generator='generator.h5',
    models_saved_path='./models/',
    latent_mean_distribution=0.0,
    latent_stander_deviation=1.0,
    smoothing_rate=0.1
)

# Compile and train
adversarial_algorithm.compile(
    Adam(learning_rate=0.0002, beta_1=0.5, beta_2=0.9),
    Adam(learning_rate=0.0002, beta_1=0.5, beta_2=0.9),
    BinaryCrossentropy(), BinaryCrossentropy()
)

adversarial_algorithm.fit(
    x_real_samples,
    to_categorical(y_real_samples, num_classes=3),
    epochs=100, batch_size=32
)

# Generate synthetic samples
samples = adversarial_algorithm.get_samples(number_samples_per_class)

🌊 Denoising Diffusion Model

State-of-the-art generation using progressive denoising diffusion processes. Achieves high-quality samples through iterative refinement.

State-of-the-art High Quality Advanced 
import numpy
import tensorflow
from tensorflow.keras.optimizers import Adam
from tensorflow.keras.losses import MeanSquaredError
from tensorflow.keras.utils import to_categorical
from maldatagen.engine.models.DenoisingDiffusion.DiffusionModelUnet import UNetDenoisingModel
from maldatagen.engine.algorithms.DenoisingDiffusion.AlgorithmDenoisingDiffusion import DenoisingDiffusionAlgorithm
from maldatagen.engine.algorithms.LatentDiffusion.GaussianLatentDiffusion import GaussianDiffusion

# Configuration
number_samples_per_class = {
    "classes": {0: 100, 1: 200, 2: 150},
    "number_classes": 3
}

# Initialize UNet models
first_instance_unet = UNetDenoisingModel(
    output_shape=1200,
    embedding_channels=1,
    list_neurons_per_level=[1, 2, 4],
    list_attentions=[False, True, True],
    number_residual_blocks=2,
    normalization_groups=1,
    intermediary_activation_function='swish',
    intermediary_activation_alpha=0.2,
    last_layer_activation='linear',
    number_samples_per_class=number_samples_per_class
)

first_unet_model = first_instance_unet.build_model()
second_unet_model = first_instance_unet.build_model()
second_unet_model.set_weights(first_unet_model.get_weights())

# Initialize Gaussian Diffusion
gaussian_diffusion_util = GaussianDiffusion(
    beta_start=1e-4, beta_end=0.02, time_steps=1000,
    clip_min=-1.0, clip_max=1.0
)

# Setup diffusion algorithm
denoising_diffusion_algorithm = DenoisingDiffusionAlgorithm(
    output_shape=1200,
    first_unet_model=first_unet_model,
    second_unet_model=second_unet_model,
    gdf_util=gaussian_diffusion_util,
    optimizer_autoencoder=Adam(learning_rate=0.0001),
    optimizer_diffusion=Adam(learning_rate=0.0001),
    time_steps=1000, ema=0.9999, margin=1.0
)

# Train and generate
denoising_diffusion_algorithm.compile(
    loss=MeanSquaredError(), optimizer=Adam(learning_rate=0.0001)
)
x_real_samples = tensorflow.expand_dims(x_real_samples, axis=-1)
denoising_diffusion_algorithm.fit(
    x_real_samples, to_categorical(y_real_samples, num_classes=3),
    epochs=100, batch_size=32
)
samples = denoising_diffusion_algorithm.get_samples(number_samples_per_class)

⚡ Wasserstein GAN (WGAN)

Improved GAN training stability using Wasserstein distance. Addresses mode collapse and provides meaningful loss metrics.

Stable Training Wasserstein Distance No Mode Collapse 
import numpy
import tensorflow
from tensorflow.keras.optimizers import Adam
from tensorflow.keras.utils import to_categorical
from maldatagen.engine.models.Wasserstein.ModelWassersteinGAN import WassersteinModel
from maldatagen.engine.algorithms.Wasserstein.AlgorithmWassersteinGAN import WassersteinAlgorithm

# Configuration
number_samples_per_class = {
    "classes": {0: 100, 1: 200, 2: 150},
    "number_classes": 3
}

# Initialize WGAN
wasserstein_model = WassersteinModel(
    latent_dimension=100,
    output_shape=1200,
    activation_function='leakyrelu',
    initializer_mean=0.0,
    initializer_deviation=0.02,
    dropout_decay_rate_g=0.3,
    dropout_decay_rate_d=0.4,
    last_layer_activation='tanh',
    dense_layer_sizes_g=[256, 512, 1024],
    dense_layer_sizes_d=[1024, 512, 256],
    dataset_type=numpy.float32,
    number_samples_per_class=number_samples_per_class
)

# Define loss functions
def discriminator_loss(real_img, fake_img):
    return tensorflow.reduce_mean(fake_img) - tensorflow.reduce_mean(real_img)

def generator_loss(fake_img):
    return -tensorflow.reduce_mean(fake_img)

# Setup algorithm
wasserstein_algorithm = WassersteinAlgorithm(
    generator_model=wasserstein_model.get_generator(),
    discriminator_model=wasserstein_model.get_discriminator(),
    latent_dimension=100,
    generator_loss_fn='wasserstein',
    discriminator_loss_fn='wasserstein',
    file_name_discriminator='wgan_discriminator.h5',
    file_name_generator='wgan_generator.h5',
    models_saved_path='./models/',
    latent_mean_distribution=0.0,
    latent_standard_deviation=1.0,
    smoothing_rate=0.1,
    discriminator_steps=5,
    clip_value=0.01
)

# Train and generate
wasserstein_algorithm.compile(
    Adam(learning_rate=0.0002, beta_1=0.5, beta_2=0.9),
    Adam(learning_rate=0.0002, beta_1=0.5, beta_2=0.9),
    generator_loss, discriminator_loss
)
wasserstein_algorithm.fit(
    x_real_samples, to_categorical(y_real_samples, num_classes=3),
    epochs=100, batch_size=32
)
samples = wasserstein_algorithm.get_samples(number_samples_per_class)

Evaluation Metrics

📈 Binary Classification Metrics

Comprehensive binary metrics including accuracy, precision, recall, F1-score, and more for model evaluation.

Classification Model Evaluation Performance Metrics 
from dataclasses import dataclass
from maldatagen.Engine.Metrics.MetricsManager import Metrics

# Define metrics configuration
@dataclass
class MetricsConfig:
    classifier: list = None
    number_k_folds: int = 5

    def __post_init__(self):
        # Define all available classifiers
        self.dictionary_classifiers_name = [
            'RandomForest', 'DecisionTree', 'SVM',
            'KNN', 'LogisticRegression'
        ]
        if self.classifier is None:
            self.classifier = ['RandomForest', 'SVM', 'KNN']

# Initialize metrics manager
config = MetricsConfig(
    classifier=['RandomForest', 'DecisionTree', 'LogisticRegression'],
    number_k_folds=5
)

metrics_manager = Metrics(config)

# Calculate binary metrics for TS-TR evaluation
# (Train on Synthetic, Test on Real)
real_labels = [0, 1, 0, 1, 1, 0, 1, 0]
predicted_labels = [0, 1, 1, 1, 1, 0, 0, 0]

metrics_manager.get_binary_metrics(
    real_labels=real_labels,
    predict_labels=predicted_labels,
    evaluation_type='TS-TR',
    classifier='RandomForest',
    fold=1
)

# Calculate binary metrics for TR-TS evaluation
# (Train on Real, Test on Synthetic)
metrics_manager.get_binary_metrics(
    real_labels=real_labels,
    predict_labels=predicted_labels,
    evaluation_type='TR-TS',
    classifier='RandomForest',
    fold=1
)

# Access calculated metrics
accuracy = metrics_manager._dictionary_metrics['TS-TR']['RandomForest']['1-Fold']['Accuracy']
precision = metrics_manager._dictionary_metrics['TS-TR']['RandomForest']['1-Fold']['Precision']
recall = metrics_manager._dictionary_metrics['TS-TR']['RandomForest']['1-Fold']['Recall']
f1_score = metrics_manager._dictionary_metrics['TS-TR']['RandomForest']['1-Fold']['F1Score']

print(f"Accuracy: {accuracy:.4f}")
print(f"Precision: {precision:.4f}")
print(f"Recall: {recall:.4f}")
print(f"F1-Score: {f1_score:.4f}")

@dataclass

    number_k_folds=5
)

# Initialize metrics manager
metrics = Metrics(config)

# Simulate k-fold cross-validation
for fold in range(5):
    print(f"\nProcessing Fold {fold + 1}/5")

    # Generate sample data (replace with your actual data)
    x_real = np.random.randn(1000, 50)
    y_real = np.random.randint(0, 2, 1000)
    x_synthetic = np.random.randn(1000, 50)
    y_synthetic = np.random.randint(0, 2, 1000)

    # Start monitoring
    metrics.monitoring_start_training()

    # TS-TR Evaluation: Train on Synthetic, Test on Real
    clf_tstr = RandomForestClassifier(n_estimators=100, random_state=42)
    clf_tstr.fit(x_synthetic, y_synthetic)
    y_pred_tstr = clf_tstr.predict(x_real)

    metrics.get_binary_metrics(
        real_labels=y_real,
        predict_labels=y_pred_tstr,
        evaluation_type='TS-TR',
        classifier='RandomForest',
        fold=fold + 1
    )

    # TR-TS Evaluation: Train on Real, Test on Synthetic
    clf_trts = RandomForestClassifier(n_estimators=100, random_state=42)
    clf_trts.fit(x_real, y_real)
    y_pred_trts = clf_trts.predict(x_synthetic)

    metrics.get_binary_metrics(
        real_labels=y_synthetic,
        predict_labels=y_pred_trts,
        evaluation_type='TR-TS',
        classifier='RandomForest',
        fold=fold + 1
    )

    metrics.monitoring_stop_training(fold)

    # Calculate distance metrics
    metrics.monitoring_start_generating()

    metrics.get_distance_metrics(
        x_evaluation_real=x_real,
        x_evaluation_synthetic=x_synthetic,
        evaluation_type='R-S',
        fold=fold + 1
    )

    metrics.monitoring_stop_generating(fold)

    print(f"  ✓ Fold {fold + 1} completed")
